// 🛠️ FD6.4 終極修正：直接繪製原始 Video 影格，徹底排除 UI/魔法陣圖層干擾 function processLuminancePeakTracking() { if (!video.videoWidth || video.readyState < 2) return; let sampleW = 320; let sampleH = Math.round(320 * (window.innerHeight / window.innerWidth)); sampleCanvas.width = sampleW; sampleCanvas.height = sampleH; sampleCtx.save(); // 鏡像翻轉以對齊畫面 sampleCtx.translate(sampleW, 0); sampleCtx.scale(-1, 1); let vW = video.videoWidth; let vH = video.videoHeight; let scale = Math.max(sampleW / vW, sampleH / vH); let rW = vW * scale; let rH = vH * scale; let offX = (rW - sampleW) / 2; let offY = (rH - sampleH) / 2; // 【核心修正】直接繪製相機 video 影格，不被 Canvas 上繪製的魔法陣與文字影響！ sampleCtx.drawImage(video, -offX, -offY, rW, rH); sampleCtx.restore(); let imgData = sampleCtx.getImageData(0, 0, sampleW, sampleH); let data = imgData.data; let darkPixels = []; let maxGlobalLum = 0; let globalBrightX = -1, globalBrightY = -1; for (let y = 0; y < sampleH; y += 2) { for (let x = 0; x < sampleW; x += 2) { let idx = (y * sampleW + x) * 4; let r = data[idx], g = data[idx + 1], b = data[idx + 2]; let maxC = Math.max(r, g, b); let minC = Math.min(r, g, b); let lum = 0.299 * r + 0.587 * g + 0.114 * b; // 避開臉部 if (userHatAR.faceBounds) { let relX = x / sampleW; let relY = y / sampleH; if (relX > userHatAR.faceBounds.left && relX < userHatAR.faceBounds.right && relY > userHatAR.faceBounds.top && relY < userHatAR.faceBounds.bottom) { continue; } } // 1. 全螢幕備援最高亮度點 (降低極限門檻至 150 以適應鏡頭壓光) if (lum > maxGlobalLum) { maxGlobalLum = lum; globalBrightX = x; globalBrightY = y; } // 2. 黑棒特徵 (寬鬆度放寬至 maxC < 110) if (maxC < 110 && (maxC - minC) < 40) { darkPixels.push({ x: x, y: y }); } } } let trackedX = -1, trackedY = -1; // 路徑 A：抓黑棒最頂端 if (darkPixels.length > 10) { let minY = sampleH; let tipPoint = null; darkPixels.forEach(p => { if (p.y < minY) { minY = p.y; tipPoint = p; } }); if (tipPoint) { let bestLightX = tipPoint.x; let bestLightY = tipPoint.y; let maxTipLum = 0; for (let dy = -25; dy <= 5; dy++) { for (let dx = -15; dx <= 15; dx++) { let inspectX = Math.min(Math.max(0, tipPoint.x + dx), sampleW - 1); let inspectY = Math.min(Math.max(0, tipPoint.y + dy), sampleH - 1); let idx = (inspectY * sampleW + inspectX) * 4; let l = 0.299 * data[idx] + 0.587 * data[idx+1] + 0.114 * data[idx+2]; if (l > maxTipLum) { maxTipLum = l; bestLightX = inspectX; bestLightY = inspectY; } } } trackedX = bestLightX; trackedY = bestLightY; } } // 路徑 B：黑棒消失時直接強制鎖定亮點 if (trackedX === -1 && maxGlobalLum > 140) { trackedX = globalBrightX; trackedY = globalBrightY; } if (trackedX !== -1 && trackedY !== -1) { crosshair.hasTracking = true; crosshair.targetX = trackedX * (canvas.width / sampleW); crosshair.targetY = trackedY * (canvas.height / sampleH); if (gameState === 'drawing') { checkWandPointCollision(crosshair.targetX, crosshair.targetY); } if (gameState === 'ready') { let btnRect = startBtn.getBoundingClientRect(); if (crosshair.targetX > btnRect.left && crosshair.targetX < btnRect.right && crosshair.targetY > btnRect.top && crosshair.targetY < btnRect.bottom) { startCountdownFlow(); } } } else { crosshair.hasTracking = false; } }
