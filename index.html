// 🛠️ FD6.3 重構：雙軌判定（寬鬆黑棒頂點 + 獨立高亮保險） function processLuminancePeakTracking() { if (!video.videoWidth || video.readyState < 2) return; let sampleW = 320; let sampleH = Math.round(320 * (window.innerHeight / window.innerWidth)); sampleCanvas.width = sampleW; sampleCanvas.height = sampleH; sampleCtx.save(); sampleCtx.translate(sampleW, 0); sampleCtx.scale(-1, 1); let vW = video.videoWidth; let vH = video.videoHeight; let scale = Math.max(sampleW / vW, sampleH / vH); let rW = vW * scale; let rH = vH * scale; let offX = (rW - sampleW) / 2; let offY = (rH - sampleH) / 2; sampleCtx.drawImage(video, -offX, -offY, rW, rH); sampleCtx.restore(); let imgData = sampleCtx.getImageData(0, 0, sampleW, sampleH); let data = imgData.data; let darkPixels = []; let maxGlobalLum = 0; let globalBrightX = -1, globalBrightY = -1; for (let y = 0; y < sampleH; y += 2) { for (let x = 0; x < sampleW; x += 2) { let idx = (y * sampleW + x) * 4; let r = data[idx], g = data[idx + 1], b = data[idx + 2]; let maxC = Math.max(r, g, b); let minC = Math.min(r, g, b); let lum = 0.299 * r + 0.587 * g + 0.114 * b; // 避開臉部 if (userHatAR.faceBounds) { let relX = x / sampleW; let relY = y / sampleH; if (relX > userHatAR.faceBounds.left && relX < userHatAR.faceBounds.right && relY > userHatAR.faceBounds.top && relY < userHatAR.faceBounds.bottom) { continue; } } // 1. 收集全畫面的最高亮度點（作為備援保險） if (lum > maxGlobalLum) { maxGlobalLum = lum; globalBrightX = x; globalBrightY = y; } // 2. 寬鬆黑棒特徵：亮度 < 100 且 彩度低 (maxC - minC < 35) if (maxC < 100 && (maxC - minC) < 35) { darkPixels.push({ x: x, y: y }); } } } let trackedX = -1, trackedY = -1; // 【路徑 A】：順利抓到黑棒管身 if (darkPixels.length > 15) { let minY = sampleH; let tipPoint = null; darkPixels.forEach(p => { if (p.y < minY) { minY = p.y; tipPoint = p; } }); if (tipPoint) { let bestLightX = tipPoint.x; let bestLightY = tipPoint.y; let maxTipLum = 0; // 在黑棒頂端向上 20px 範圍內尋找 LED 亮點 for (let dy = -20; dy <= 5; dy++) { for (let dx = -12; dx <= 12; dx++) { let inspectX = Math.min(Math.max(0, tipPoint.x + dx), sampleW - 1); let inspectY = Math.min(Math.max(0, tipPoint.y + dy), sampleH - 1); let idx = (inspectY * sampleW + inspectX) * 4; let l = 0.299 * data[idx] + 0.587 * data[idx+1] + 0.114 * data[idx+2]; if (l > maxTipLum) { maxTipLum = l; bestLightX = inspectX; bestLightY = inspectY; } } } trackedX = bestLightX; trackedY = bestLightY; } } // 【路徑 B（保險）】：黑棒沒抓滿時，直接採用區域外的全螢幕高亮 LED 點 if (trackedX === -1 && maxGlobalLum > 180) { trackedX = globalBrightX; trackedY = globalBrightY; } // 順利鎖定座標 if (trackedX !== -1 && trackedY !== -1) { crosshair.hasTracking = true; crosshair.targetX = trackedX * (canvas.width / sampleW); crosshair.targetY = trackedY * (canvas.height / sampleH); if (gameState === 'drawing') { checkWandPointCollision(crosshair.targetX, crosshair.targetY); } if (gameState === 'ready') { let btnRect = startBtn.getBoundingClientRect(); if (crosshair.targetX > btnRect.left && crosshair.targetX < btnRect.right && crosshair.targetY > btnRect.top && crosshair.targetY < btnRect.bottom) { startCountdownFlow(); } } } else { crosshair.hasTracking = false; } }
