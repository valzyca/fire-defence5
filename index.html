// 🛠️ FD6.2 重構：黑色魔杖長棒 + 頂端發光點 雙特徵幾何檢測 function processLuminancePeakTracking() { if (!video.videoWidth || video.readyState < 2) return; let sampleW = 320; let sampleH = Math.round(320 * (window.innerHeight / window.innerWidth)); sampleCanvas.width = sampleW; sampleCanvas.height = sampleH; sampleCtx.save(); sampleCtx.translate(sampleW, 0); sampleCtx.scale(-1, 1); let vW = video.videoWidth; let vH = video.videoHeight; let scale = Math.max(sampleW / vW, sampleH / vH); let rW = vW * scale; let rH = vH * scale; let offX = (rW - sampleW) / 2; let offY = (rH - sampleH) / 2; sampleCtx.drawImage(video, -offX, -offY, rW, rH); sampleCtx.restore(); let imgData = sampleCtx.getImageData(0, 0, sampleW, sampleH); let data = imgData.data; // 階段 1：收集所有符合「黑色棒體」特徵的像素區域 let darkPixels = []; for (let y = 0; y < sampleH; y += 2) { for (let x = 0; x < sampleW; x += 2) { let idx = (y * sampleW + x) * 4; let r = data[idx], g = data[idx + 1], b = data[idx + 2]; // 計算 RGB 的極大值與極小值以估算亮度 Value let maxC = Math.max(r, g, b); let minC = Math.min(r, g, b); // 避開臉部 if (userHatAR.faceBounds) { let relX = x / sampleW; let relY = y / sampleH; if (relX > userHatAR.faceBounds.left && relX < userHatAR.faceBounds.right && relY > userHatAR.faceBounds.top && relY < userHatAR.faceBounds.bottom) { continue; } } // 條件：黑色棒體（RGB 最大值低於 65，且 RGB 差異小呈暗灰色/黑色） if (maxC < 65 && (maxC - minC) < 25) { darkPixels.push({ x: x, y: y }); } } } // 階段 2：當找到足夠多的黑色像素時，擬合出長棒的位置與頂端 if (darkPixels.length > 30) { // 計算黑色區域重心 let sumX = 0, sumY = 0; darkPixels.forEach(p => { sumX += p.x; sumY += p.y; }); let cX = sumX / darkPixels.length; let cY = sumY / darkPixels.length; // 沿著 Y 軸（上方）尋找黑色區域最頂端/最高點 let minY = sampleH; let tipPoint = null; darkPixels.forEach(p => { if (p.y < minY) { minY = p.y; tipPoint = p; } }); if (tipPoint) { // 在頂端點周圍 ±15 像素範圍內，精確尋找「發光亮點」 let bestLightX = tipPoint.x; let bestLightY = tipPoint.y; let maxBrightness = 0; for (let dy = -15; dy <= 5; dy++) { for (let dx = -10; dx <= 10; dx++) { let inspectX = Math.min(Math.max(0, tipPoint.x + dx), sampleW - 1); let inspectY = Math.min(Math.max(0, tipPoint.y + dy), sampleH - 1); let idx = (inspectY * sampleW + inspectX) * 4; let brightness = 0.299 * data[idx] + 0.587 * data[idx+1] + 0.114 * data[idx+2]; if (brightness > maxBrightness) { maxBrightness = brightness; bestLightX = inspectX; bestLightY = inspectY; } } } // 鎖定準星座標 crosshair.hasTracking = true; crosshair.targetX = bestLightX * (canvas.width / sampleW); crosshair.targetY = bestLightY * (canvas.height / sampleH); if (gameState === 'drawing') { checkWandPointCollision(crosshair.targetX, crosshair.targetY); } if (gameState === 'ready') { let btnRect = startBtn.getBoundingClientRect(); if (crosshair.targetX > btnRect.left && crosshair.targetX < btnRect.right && crosshair.targetY > btnRect.top && crosshair.targetY < btnRect.bottom) { startCountdownFlow(); } } return; } } crosshair.hasTracking = false; }
