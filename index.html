// src/ui/avatarCanvas.js - 魔法捏臉繪製引擎 (by Gemini) export class AvatarEngine { constructor(canvasId) { this.canvas = document.getElementById(canvasId); this.ctx = this.canvas.getContext('2d'); // 預設捏臉狀態數據 (JSON 可讀寫) this.avatarState = { skinColor: '#FFE0BD', // 膚色 hatType: 'wizard_hat', // 帽子類型 hatColor: '#0B132B', // 帽子顏色 eyeType: 'star_eyes', // 眼睛類型 eyeColor: '#48E5C2', // 眼睛顏色 (Tiffany綠) expression: 'smile', // 表情 accessory: 'glasses' // 配件 }; } // 設定部位參數 (當玩家點擊 UI 切換時調用) updateFeature(feature, value) { this.avatarState[feature] = value; this.render(); // 即時重新繪製 } // 繪製學徒頭像核心邏輯 render() { const ctx = this.ctx; ctx.clearRect(0, 0, this.canvas.width, this.canvas.height); // 1. 繪製背景光暈 ctx.fillStyle = '#1A2640'; ctx.fillRect(0, 0, 200, 200); // 2. 繪製臉型 (底座) ctx.fillStyle = this.avatarState.skinColor; ctx.beginPath(); ctx.arc(100, 110, 45, 0, Math.PI * 2); ctx.fill(); // 3. 繪製眼睛 (根據類型動態繪製) ctx.fillStyle = this.avatarState.eyeColor; if (this.avatarState.eyeType === 'star_eyes') { // 繪製閃亮星星眼 ctx.font = '20px Arial'; ctx.fillText('★', 75, 110); ctx.fillText('★', 110, 110); } else { // 圓圓大眼 ctx.beginPath(); ctx.arc(80, 105, 7, 0, Math.PI * 2); ctx.arc(120, 105, 7, 0, Math.PI * 2); ctx.fill(); } // 4. 繪製魔法巫師帽 ctx.fillStyle = this.avatarState.hatColor; ctx.beginPath(); ctx.moveTo(40, 90); ctx.lineTo(100, 20); // 帽頂 ctx.lineTo(160, 90); ctx.closePath(); ctx.fill(); // 帽子飾條 (Tiffany綠) ctx.fillStyle = '#48E5C2'; ctx.fillRect(50, 80, 100, 8); } // 轉換為 16-Bit 復古像素風並導出圖片 (給學徒證使用) exportToPixelCard() { // 將 200x200 的 Canvas 縮小至 32x32，再放大實現高品質 Pixel Art 效果！ const offscreenCanvas = document.createElement('canvas'); offscreenCanvas.width = 32; offscreenCanvas.height = 32; const offCtx = offscreenCanvas.getContext('2d'); // 禁用平滑化，保留硬邊像素感 offCtx.imageSmoothingEnabled = false; offCtx.drawImage(this.canvas, 0, 0, 32, 32); return offscreenCanvas.toDataURL(); // 回傳 16-bit 像素頭像字串 } }
