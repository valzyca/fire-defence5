// 🛠️ FD6.5 終極重構：專利綠色 LED 色彩鎖定 (Green LED Targeted Tracker) function processLuminancePeakTracking() { if (!video.videoWidth || video.readyState < 2) return; let sampleW = 320; let sampleH = Math.round(320 * (window.innerHeight / window.innerWidth)); sampleCanvas.width = sampleW; sampleCanvas.height = sampleH; sampleCtx.save(); sampleCtx.translate(sampleW, 0); sampleCtx.scale(-1, 1); let vW = video.videoWidth; let vH = video.videoHeight; let scale = Math.max(sampleW / vW, sampleH / vH); let rW = vW * scale; let rH = vH * scale; let offX = (rW - sampleW) / 2; let offY = (rH - sampleH) / 2; // 直接抓原始 video 鏡頭影格 sampleCtx.drawImage(video, -offX, -offY, rW, rH); sampleCtx.restore(); let imgData = sampleCtx.getImageData(0, 0, sampleW, sampleH); let data = imgData.data; let greenSumX = 0, greenSumY = 0, greenCount = 0; let maxGreenValue = 0; let bestX = -1, bestY = -1; for (let y = 0; y < sampleH; y += 2) { for (let x = 0; x < sampleW; x += 2) { let idx = (y * sampleW + x) * 4; let r = data[idx], g = data[idx + 1], b = data[idx + 2]; // 綠色 LED 核心特徵：綠色 (G) 必須顯著高於 紅(R) 與 藍(B)，且本身足夠亮 // 這能 100% 過濾天花板白燈、臉部膚色與背景黑櫃子！ if (g > 90 && g > r * 1.2 && g > b * 1.2) { greenSumX += x; greenSumY += y; greenCount++; if (g > maxGreenValue) { maxGreenValue = g; bestX = x; bestY = y; } } } } let trackedX = -1, trackedY = -1; if (greenCount > 0) { // 取得綠色光點區域的質心（Centroid），避免單像素抖動 trackedX = greenSumX / greenCount; trackedY = greenSumY / greenCount; } else if (bestX !== -1) { trackedX = bestX; trackedY = bestY; } // 順利鎖定綠色發光魔杖頂端 if (trackedX !== -1 && trackedY !== -1) { crosshair.hasTracking = true; crosshair.targetX = trackedX * (canvas.width / sampleW); crosshair.targetY = trackedY * (canvas.height / sampleH); if (gameState === 'drawing') { checkWandPointCollision(crosshair.targetX, crosshair.targetY); } if (gameState === 'ready') { let btnRect = startBtn.getBoundingClientRect(); if (crosshair.targetX > btnRect.left && crosshair.targetX < btnRect.right && crosshair.targetY > btnRect.top && crosshair.targetY < btnRect.bottom) { startCountdownFlow(); } } } else { crosshair.hasTracking = false; } }
